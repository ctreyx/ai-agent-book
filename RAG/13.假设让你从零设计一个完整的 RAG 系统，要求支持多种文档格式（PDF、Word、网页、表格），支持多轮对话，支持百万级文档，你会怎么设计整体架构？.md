> **第一层：数据处理层（Data Pipeline）**
>
> ```python
> # 不同格式用不同的 Loader
> from langchain_community.document_loaders import (
>     PyPDFLoader,           # PDF
>     Docx2txtLoader,        # Word
>     WebBaseLoader,         # 网页
>     CSVLoader,             # CSV 表格
>     UnstructuredExcelLoader # Excel
> )
> 
> # 统一处理流程
> def process_document(file_path: str):
>     # 1. 根据文件类型选择 loader
>     loader = get_loader_by_type(file_path)
>     raw_docs = loader.load()
>     
>     # 2. 数据清洗（去除页眉页脚、特殊字符）
>     cleaned = clean_documents(raw_docs)
>     
>     # 3. 表格检测：如果包含表格，走特殊处理
>     tables, texts = separate_tables_and_text(cleaned)
>     
>     # 4a. 普通文本 → 语义切片
>     text_chunks = semantic_splitter.split(texts)
>     
>     # 4b. 表格 → 转自然语言 + 存SQL
>     table_chunks = table_to_natural_language(tables)
>     save_to_sql(tables)  # Text2SQL 用
>     
>     # 5. 统一 Embedding → 存入向量库
>     all_chunks = text_chunks + table_chunks
>     vectorstore.add_documents(all_chunks)
> ```
>
> | 文档格式 | Loader | 特殊处理 |
> |---|---|---|
> | PDF | PyPDFLoader | 检测表格区域单独处理 |
> | Word | Docx2txtLoader | 保留标题层级结构 |
> | 网页 | WebBaseLoader | 去除导航栏、广告等噪声 |
> | Excel/CSV | CSVLoader | 每行转自然语言 + 存 SQL |
> | Markdown | TextLoader | 按标题切片 |

> **第二层：存储层**
>
> ```
> 百万级文档的存储方案：
> 
> ┌─────────────────────────────────────────┐
> │           PostgreSQL                     │
> │  ├── pgvector 扩展 → 向量检索            │
> │  ├── 普通表       → 结构化数据/表格       │
> │  └── 全文索引     → BM25 关键词检索       │
> └─────────────────────────────────────────┘
>            +
> ┌─────────────────────────────────────────┐
> │           Redis                          │
> │  ├── 热门查询缓存                        │
> │  ├── 对话历史缓存                        │
> │  └── Embedding 结果缓存                  │
> └─────────────────────────────────────────┘
> ```
>
> **为什么选 PostgreSQL 一体化：**
> - pgvector 做向量检索，省去独立向量库的运维
> - 普通 SQL 表做 Text2SQL 查询
> - 全文索引做 BM25 检索
> - 一个数据库搞定三件事，运维成本最低
> - 如果规模超过千万级再考虑迁移到 Milvus
>
> **百万级必须加缓存：**
>
> ```python
> import redis
> import hashlib
> 
> r = redis.Redis()
> 
> def cached_retrieval(question: str):
>     # 用问题的 hash 做缓存 key
>     cache_key = f"rag:{hashlib.md5(question.encode()).hexdigest()}"
>     
>     # 先查缓存
>     cached = r.get(cache_key)
>     if cached:
>         return json.loads(cached)  # 命中缓存，直接返回
>     
>     # 未命中 → 正常检索
>     result = full_rag_pipeline(question)
>     
>     # 写入缓存，1小时过期
>     r.setex(cache_key, 3600, json.dumps(result))
>     return result
> ```

> **第五层：接口层 + 对话管理**
>
> ```python
> from fastapi import FastAPI, WebSocket
> 
> app = FastAPI()
> 
> # 多轮对话管理
> class ConversationManager:
>     def __init__(self):
>         self.sessions = {}  # session_id → 对话历史
>     
>     def get_history(self, session_id: str):
>         history = self.sessions.get(session_id, [])
>         # 超过10轮 → 用 LLM 压缩摘要
>         if len(history) > 10:
>             history = self.summarize(history)
>         return history
>     
>     def summarize(self, history):
>         """用 LLM 把长历史压缩成摘要"""
>         prompt = f"把以下对话历史压缩成200字以内的摘要：\n{history}"
>         summary = llm.invoke(prompt).content
>         return [{"role": "system", "content": f"之前的对话摘要：{summary}"}]
> 
> # 流式输出接口（用户体验好）
> @app.websocket("/ws/chat")
> async def chat(websocket: WebSocket):
>     await websocket.accept()
>     session_id = create_session()
>     
>     while True:
>         question = await websocket.receive_text()
>         history = conversation_manager.get_history(session_id)
>         
>         async for chunk in rag_pipeline.astream(question, history):
>             await websocket.send_text(chunk)
> ```
