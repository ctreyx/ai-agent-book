第二层：Pydantic 结构化校验

Python

from pydantic import BaseModel, Field
from typing import Optional

class LocationInfo(BaseModel):
    name: str
    longitude: float
    latitude: float
    zoom: int = 16

class AgentResponse(BaseModel):
    answer: str = Field(description="回答内容")
    intent: str = Field(description="chat/locate/plan")
    location: Optional[LocationInfo] = None
    suggest_jump: bool = False
    sources: list[str] = []

def parse_response(raw_text: str) -> AgentResponse:
    try:
        data = json.loads(raw_text)
        return AgentResponse(**data)    # 校验通过
    except (json.JSONDecodeError, ValidationError) as e:
        raise InvalidFormatError(e)     # 校验失败
第三层：重试 + 降级

Python

async def safe_agent_call(question: str, max_retries: int = 2):
    for attempt in range(max_retries + 1):
        raw = await llm.ainvoke(question)
        
        try:
            result = parse_response(raw.content)
            
            # 额外校验：坐标必须在绍兴范围内
            if result.location:
                if not (120.3 < result.location.longitude < 120.8
                    and 29.7 < result.location.latitude < 30.2):
                    raise ValueError("坐标超出绍兴范围")
            
            return result
        
        except Exception as e:
            if attempt < max_retries:
                continue    # 重试
            else:
                # 全部失败 → 降级返回纯文本
                return AgentResponse(
                    answer=raw.content,
                    intent="chat",
                    location=None,
                    suggest_jump=False,
                    sources=[]
                )
降级策略：JSON 解析全部失败时，把 LLM 原始回答当作纯文本返回，前端当普通对话展示，不做地图跳转。用户体验不会中断。
